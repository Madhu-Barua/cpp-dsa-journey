int rear = -1, front = -1;
int priorities[5];
char values[5][30];

int isFull() {
    if (rear == 4) {
        printf("Queue is full");

    }
}

int isEmpty() {

void enqueue(int priority, char value[30]) {
    if (rear == 4) {
        printf("Queue is full");
    } else {
        rear = rear + 1;
        priorities[rear] = priority;
        strcpy(values[rear],value);
    }
}

int getHighestPriority() {
    int i, priority;
    priority = priorities[0];
    for (i = 1; i <= rear; i++) {
        if (priorities[i] > priority) {
            priority = priorities[i];
        }
    }
    return priority;
}

    return value;
}
