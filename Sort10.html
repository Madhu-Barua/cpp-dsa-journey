int partition(int A[], int low, int high)
{
    int pivot = A[low];
    int start = low;
    int end = high;
    int temp;

    while (start < end)
    {
        while (A[start] <= pivot)
        {
            start++;
        }

        while (A[end] > pivot)
        {
            end--;
        }

        if (start < end)
        {
            temp = A[start];
            A[start] = A[end];
            A[end] = temp;
        }
    } 
    temp = A[low];
    A[low] = A[end];
    A[end] = temp;
    return end;
}

void quickSort(int A[], int low, int high)
{
    int loc;

    if (low < high)
    {
        loc = partition(A, low, high); 
        quickSort(A, low, loc - 1);
        quickSort(A, loc + 1, high);
    }
}

int main()
{   
    int size;
    printf("Enter the size: ");
    scanf("%d", &size);
 
    int A[size];
    printf("Enter the elements of array: ");
    for (int i = 0; i < size; i++)
    {
        scanf("%d", &A[i]);
    }
    quickSort(A, 0, size - 1);
    
    printf("The sorted array is: ");
    for (int i = 0; i < size; i++)
    {
        printf("%d ", A[i]);
    }
    printf("\n");
    return 0;
}
