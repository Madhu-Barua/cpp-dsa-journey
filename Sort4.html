#include< stdio.h >
void bubblesort(int a[],int size){
    if(size==0)
    return;
    else{
    for(int i=0;i < size;i++){
          if(a[i] > a[i+1]){
            int temp=a[i];
            a[i]=a[i+1];
            a[i+1]=temp;
          }  
        }
    bubblesort(a,size-1);
}}
void main(){
    int a[10],i,size;
    printf("Enter number of elements 
    you want : ");
    scanf("%d",&size);
    printf("Enter elements of array : ");
    for(i=0;i < size;i++){
        scanf("%d",&a[i]);
    }
    bubblesort(a,size-1);
    printf("Sorted Array : ");

}
