#include< stdlib.h >
int rear = -1, front = -1;
int priorities[5];
int values[5];

int isFull() {
    if (rear == 4) {
        printf("Queue is full");
    } else {
        printf("Queue is not full");
    }
}

int isEmpty() {
    if (front == -1 && rear==-1) 
        printf("Queue is Empty");
    else 
        printf("Queue is not Empty");

}

void enqueue(int priority, int value) {
    if (rear == 4) {
        printf("Queue is full");
    } else {
        rear = rear + 1;
        priorities[rear] = priority;
        values[rear] = value;
    }
}

int getHighestPriority() {
    int i, priority;
    priority = priorities[0];
    for (i = 1; i <= rear; i++) {
        if (priorities[i] > priority) {
            priority = priorities[i];
        }
    }
    return priority;
}

int delete() {
    int i, j, value, priority;
    priority = getHighestPriority();
    for (i = 0; i <= rear; i++) {
    if (priorities[i] == priority) {
     value = values[i];
        break;
      }
    }
    if (i < rear) {
    for (j = i; j < rear; j++) {
    values[j] = values[j + 1];
    priorities[j] = priorities[j + 1];
        }
    }
    rear = rear - 1;
    return value;
}

void display() {
    int i;
    printf("Elements in queue are: ");
    for (i = 0; i <= rear; i++) {
    printf("\nValue is: %d\tPriority is: %d", 
    values[i], priorities[i]);
    }
}
