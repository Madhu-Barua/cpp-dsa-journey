#include< stdio.h >
#include< conio.h >
void main(){
    int i,j,temp,a[10],n;
    printf("Enter number of elements you want : ");
    scanf("%d",&n);
    printf("Enter elements of array : ");
    for(i=0;i < n;i++){
        scanf("%d",&a[i]);
    }
    for(i=0;i < n-1;i++){
        for(j=0;j < n-i-1;j++){
          if(a[j] > a[j+1]){
            temp=a[j];
            a[j]=a[j+1];
            a[j+1]=temp;
          }  
        }
    }
    printf("Sorted Array : \n");
    for(i=0;i < n;i++){
        printf("%d",a[i]);   
    }
    printf("\n");
}
