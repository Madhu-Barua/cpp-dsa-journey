#include< stdio.h >
#include< stdlib.h >
void main(){ 
    int n,i,loc=0;
    char item;
    printf("Enter size of string : ");
    scanf("%d",&n);
    char a[n];
    printf("Enter string : ");
    scanf("%s",a);
    printf("Enter element to search : ");
    fflush(stdin);
    scanf("%c",&item);
    printf("Linear Search");
    printf("\n");
    for(i=0;a[i]!='\0';i++){
        if(a[i]==item){
            loc=i+1;
            break;
        }
    }
    if(loc>0){
        printf("Search Successful, %c 
    Element  found at %d pos",item,loc);
        printf("\n");}
    else{
        printf("Search Unsuccessful\n");
    }
}
