#include< stdio.h >
#include< conio.h >
void main(){
    int a[5],n,i,loc=0,item;
    printf("How many elements : ");
    scanf("%d",&n);
    printf("Enter Array : ");
    for(i=0;i < n;i++){
        scanf("%d",&a[i]);
    }
    printf("Enter element to search : ");
    scanf("%d",&item);
    printf("Linear Search");
    printf("\n");
    for(i=0;i < n;i++){
        if(a[i]==item){
            loc=i+1;
            break;
        }
    }
    if(loc>0){
        printf("Search Successful, Element 
       found at %d pos",loc);
        printf("\n");}
    else{
        printf("Search Unsuccessful\n");
    }
}
