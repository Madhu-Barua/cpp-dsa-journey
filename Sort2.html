#include< stdio.h >
void selectionsort(int a[],int beginindex,
int size){
    if(beginindex >= size-1){
    return;
    }
    else{
    int minindex=beginindex;
    for(int i = beginindex; i < size;i++){
    if(a[i] < a[minindex]){
      minindex=i;
          }  
        }
    int temp=a[minindex];
    a[minindex]=a[beginindex];
    a[beginindex]=temp;
    selectionsort(a,beginindex+1,size);
}}

    int i,size;
    printf("Enter number of elements
