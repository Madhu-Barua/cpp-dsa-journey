int push(char *arr,int n,int top){
    if(top==n){
        return -2;
    }
    else{
        top++;
        arr[top]='(';
        return top;
    }
}
int pop(char *arr,int n,int top){
    if(top==-1){
        return -2;
    }
    else{
        top--;
        return top;
    }
}
