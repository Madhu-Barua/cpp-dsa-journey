#include< stdio.h >
#include< string.h >
void selectionsort(char a[][30],int beginindex,
    int size){
    char temp[30];
    if(beginindex>=size-1)
    return;
    int minindex=beginindex;
    for(int i=beginindex;i < size;i++){
          if(strcmp(a[i],a[minindex])<0){
            minindex=i;
          }  
        }
   strcpy(temp,a[minindex]);
   strcpy(a[minindex],a[beginindex]);
   strcpy(a[beginindex],temp);
   selectionsort(a,beginindex+1,size);
}
void main(){
    int i,size;
    printf("Enter number of elements

    scanf("%d",&size);
    char a[size][30];
    printf("Enter elements of array : ");

    }

}
