#include< stdio.h >
int min(int arr[],int size){
    if(size==1)
    return arr[size-1];
    int possible=min(arr,size-1);
    if(possible>arr[size-1])
    return possible;
    else
    return arr[size-1];
}
int main(){
    int i,size;
    printf("Enter size : ");
    scanf("%d",&size);
    int arr[size];
    printf("Enter elements of array : ");
    for(i=0; i < size;i++){
        scanf("%d",&arr[i]);
    }
    printf("The minimum element is : %d",
    min(arr,size));
    return 0;
}
