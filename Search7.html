#include< stdio.h >
int recursion(int arr[],int low,int high,
     int a){
     if(high>=low){
        int mid=(low+high)/2;
        if(arr[mid]==a){
            return mid+1;
        }
        else if(arr[mid]>a){
    return recursion(arr,low,mid-1,a);
        }
        else{
    return recursion(arr,mid+1,high,a); 
        }
        return -1;
     }}
        
void main(){
    int n,i,a;
    printf("Enter size of array : ");
    scanf("%d",&n);
    int arr[n];
    printf("Enter elements of the array :");
    for(i=0;i < n;i++){
    scanf("%d",&arr[i]);
    }
    printf("Enter element to find : ");
    scanf("%d",&a);
    int found = recursion(arr,0,n-1,a);
    if(found == -1)
    printf("Element not found\n");
    else
    printf("%d element is found at the 
    location :%d\n",a,found);
}
