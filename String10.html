#include < stdio.h >
#include < string.h >
int main() {
    char str[100];
    printf("Enter a string: ");
    scanf("%s", str);
    int index;
    printf("Enter the index: ");
    scanf("%d", &index);
    int length = strlen(str);
    if (index >= 0 && index < length) {
        char character = str[index];
        printf("Character at index %d: %c\n", 
        index, character);
    } else {
        printf("Invalid index\n");
    return 0;
}}
