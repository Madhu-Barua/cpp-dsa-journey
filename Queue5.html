int enqueue(int *queue,int size,int item,
  int rear,int front){

  rear=front=0;
  queue[rear]=item;
  return rear;
  }

  rear=(rear+1)%size;

  return rear;
  }}

  else{
  front=front+1;
  return front;
  }}
