#include< stdio.h >
#include< string.h >
void main(){
    int i,j,n;
printf("Enter number of elements you want : ");
    scanf("%d",&n);
    char a[n][30],temp[30];
    printf("Enter elements of array : ");
    for(i=0;i < n;i++){
        scanf("%s",a[i]);
    }
    for(i=0;i < n-1;i++){
        for(j=0;j < n-i-1;j++){
          if(strcmp(a[j],a[j+1])>0){
            strcpy(temp,a[j]);
            strcpy(a[j],a[j+1]);
            strcpy(a[j+1],temp);
          }  
        }
    }
    printf("Sorted Array : \n");
    for(i=0;i < n;i++){
        printf("%s\t",a[i]);   
    }
    printf("\n");
}
