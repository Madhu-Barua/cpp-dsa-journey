#include< stdio.h >
double calculateArithmeticMean(int numbers[], 
int n, int currentIndex) {
    if (currentIndex == n) {
        return 0.0;
    }
double sumOfRemaining = calculateArithmeticMean
 (numbers, n, currentIndex + 1);
    sumOfRemaining += numbers[currentIndex];

    return sumOfRemaining;
}

int main() {
    int n;
    printf("Enter the number of elements: ");
    scanf("%d", &n);
    int numbers[n];
    printf("Enter %d numbers:\n", n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &numbers[i]);
    }
    double mean = calculateArithmeticMean
    (numbers, n, 0) / n;
    printf("Arithmetic Mean: %lf\n", mean);
    return 0;
}
